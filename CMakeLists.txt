cmake_minimum_required(VERSION 3.20)
project(femd LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# header-only core
add_library(femd INTERFACE)
target_include_directories(femd INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)

option(FEMD_BUILD_PYTHON "Build the nanobind extension" ON)
option(FEMD_WITH_FFTW "Link FFTW for the circulant (FFT) solver backend, if found" ON)
option(FEMD_OPENMP "Build the Python extension with OpenMP (macOS: brew install libomp)" OFF)

# ---- optional FFTW (Homebrew, MacPorts, system) --------------------------------
set(FEMD_FFTW_FOUND OFF)
if(FEMD_WITH_FFTW)
  find_path(FFTW3_INCLUDE_DIR fftw3.h HINTS ENV FFTW_PREFIX PATHS /opt/homebrew/include /usr/local/include /opt/local/include)
  find_library(FFTW3_LIBRARY NAMES fftw3 HINTS ENV FFTW_PREFIX PATHS /opt/homebrew/lib /usr/local/lib /opt/local/lib)
  if(FFTW3_INCLUDE_DIR AND FFTW3_LIBRARY)
    set(FEMD_FFTW_FOUND ON)
    message(STATUS "FEMd: FFTW found (${FFTW3_LIBRARY}); circulant solver enabled")
  else()
    message(STATUS "FEMd: FFTW not found; circulant solver disabled")
  endif()
endif()
option(FEMD_BUILD_TESTS  "Build the C++ tests (the Makefile is the day-to-day route)" OFF)

if(FEMD_BUILD_PYTHON)
  find_package(Python 3.10 COMPONENTS Interpreter Development.Module REQUIRED)
  execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir
                  OUTPUT_STRIP_TRAILING_WHITESPACE OUTPUT_VARIABLE nanobind_ROOT)
  find_package(nanobind CONFIG REQUIRED)
  # NOMINSIZE: nanobind's default in Release is MINSIZE, which compiles the module
  # with -Os.  That halved the speed of every kernel reached from Python (matvec,
  # assembly, solves) against the same code at -O3.
  nanobind_add_module(_femd NB_STATIC NOMINSIZE python/src/femd_bindings.cpp python/src/femd_bindings_2d.cpp)
  target_link_libraries(_femd PRIVATE femd)
  if(FEMD_FFTW_FOUND)
    target_compile_definitions(_femd PRIVATE FEMD_HAVE_FFTW)
    target_include_directories(_femd PRIVATE ${FFTW3_INCLUDE_DIR})
    target_link_libraries(_femd PRIVATE ${FFTW3_LIBRARY})
  endif()
  if(FEMD_OPENMP)
    # Apple's clang has no OpenMP of its own: point FindOpenMP at Homebrew's libomp.
    if(APPLE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
      execute_process(COMMAND brew --prefix libomp OUTPUT_VARIABLE FEMD_LIBOMP
                      OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
      if(NOT FEMD_LIBOMP)
        foreach(prefix /opt/homebrew/opt/libomp /usr/local/opt/libomp)
          if(EXISTS "${prefix}/include/omp.h")
            set(FEMD_LIBOMP "${prefix}")
          endif()
        endforeach()
      endif()
      if(NOT FEMD_LIBOMP)
        message(FATAL_ERROR "FEMD_OPENMP=ON needs libomp on macOS: brew install libomp")
      endif()
      set(OpenMP_CXX_FLAGS "-Xpreprocessor -fopenmp -I${FEMD_LIBOMP}/include")
      set(OpenMP_CXX_LIB_NAMES "omp")
      set(OpenMP_omp_LIBRARY "${FEMD_LIBOMP}/lib/libomp.dylib")
    endif()
    find_package(OpenMP REQUIRED COMPONENTS CXX)
    target_link_libraries(_femd PRIVATE OpenMP::OpenMP_CXX)
    message(STATUS "FEMd: OpenMP enabled (${OpenMP_CXX_FLAGS})")
    if(FEMD_FFTW_FOUND)
      find_library(FFTW3_OMP_LIBRARY NAMES fftw3_omp fftw3_threads HINTS ENV FFTW_PREFIX
                   PATHS /opt/homebrew/lib /usr/local/lib /opt/local/lib)
      if(FFTW3_OMP_LIBRARY)
        target_link_libraries(_femd PRIVATE ${FFTW3_OMP_LIBRARY})
        target_compile_definitions(_femd PRIVATE FEMD_FFTW_OMP)
        message(STATUS "FEMd: threaded FFTW (${FFTW3_OMP_LIBRARY})")
      endif()
    endif()
  endif()
  install(TARGETS _femd LIBRARY DESTINATION femd)
endif()

if(FEMD_BUILD_TESTS)
  file(GLOB FEMD_TESTS tests/cpp/test_*.cpp)
  foreach(t ${FEMD_TESTS})
    get_filename_component(name ${t} NAME_WE)
    if(name STREQUAL "test_smw" OR name STREQUAL "test_block_smw")
      continue()   # need FFTW; use the Makefile
    endif()
    add_executable(${name} ${t})
    target_link_libraries(${name} PRIVATE femd)
    target_include_directories(${name} PRIVATE tests/cpp)
  endforeach()
endif()
